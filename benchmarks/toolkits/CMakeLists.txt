cmake_minimum_required(VERSION 3.20)
project(byteink_toolkit_comparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui)
add_executable(qt_benchmark qt_benchmark.cpp)
target_link_libraries(qt_benchmark PRIVATE Qt6::Core Qt6::Gui)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(qt_benchmark PRIVATE -ffp-contract=off)
endif()
